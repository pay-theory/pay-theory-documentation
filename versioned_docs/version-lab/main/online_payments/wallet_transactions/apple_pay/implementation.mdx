# Apple Pay Implementation

This guide walks through implementing Apple Pay on your website using the Apple Pay JS API and Pay Theory SDK.

***

## Basic Implementation

### 1. Check Apple Pay Availability

```javascript
// Check if Apple Pay is available
if (window.ApplePaySession) {
  const merchantIdentifier = 'merchant.com.yourdomain.pay';
  
  ApplePaySession.canMakePaymentsWithActiveCard(merchantIdentifier)
    .then(canMakePayments => {
      if (canMakePayments) {
        // Show Apple Pay button
        document.getElementById('apple-pay-button').style.display = 'block';
      }
    });
}
```

### 2. Add Apple Pay Button

```html
<style>
  .apple-pay-button {
    display: none;
    -webkit-appearance: -apple-pay-button;
    -apple-pay-button-type: pay;
    -apple-pay-button-style: black;
    width: 200px;
    height: 44px;
  }
</style>

<button id="apple-pay-button" class="apple-pay-button"></button>
```

### 3. Handle Payment Request

```javascript
document.getElementById('apple-pay-button').addEventListener('click', () => {
  const request = {
    countryCode: 'US',
    currencyCode: 'USD',
    supportedNetworks: ['visa', 'masterCard', 'amex', 'discover'],
    merchantCapabilities: ['supports3DS'],
    total: {
      label: 'Your Business Name',
      amount: '10.00'
    }
  };
  
  const session = new ApplePaySession(6, request);
  
  // Handle merchant validation
  session.onvalidatemerchant = async (event) => {
    const merchantSession = await validateMerchant(event.validationURL);
    session.completeMerchantValidation(merchantSession);
  };
  
  // Handle payment authorization
  session.onpaymentauthorized = async (event) => {
    const result = await processPayment(event.payment);
    
    if (result.success) {
      session.completePayment(ApplePaySession.STATUS_SUCCESS);
    } else {
      session.completePayment(ApplePaySession.STATUS_FAILURE);
    }
  };
  
  session.begin();
});
```

## Merchant Validation

Apple Pay requires server-side merchant validation. You'll need to call your server, which then calls Pay Theory's validation endpoint.

### Server-Side Validation

```javascript
// Server endpoint (Node.js example)
app.post('/api/apple-pay/validate', async (req, res) => {
  const { validationURL } = req.body;
  
  // Call Pay Theory's merchant validation
  const response = await fetch('https://api.paytheory.com/apple-pay/validate', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${PAY_THEORY_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      validationURL: validationURL,
      merchantIdentifier: 'merchant.com.yourdomain.pay',
      displayName: 'Your Business Name',
      domainName: 'yourdomain.com'
    })
  });
  
  const merchantSession = await response.json();
  res.json(merchantSession);
});
```

### Client-Side Call

```javascript
async function validateMerchant(validationURL) {
  const response = await fetch('/api/apple-pay/validate', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ validationURL })
  });
  
  return response.json();
}
```

## Process Payment with Pay Theory

Once Apple Pay provides the payment token, process it using Pay Theory's Messenger SDK.

```javascript
async function processPayment(payment) {
  // Initialize Pay Theory Messenger
  const messenger = new PayTheoryMessenger({
    apiKey: 'YOUR_PAY_THEORY_API_KEY'
  });
  
  try {
    // Process the wallet transaction
    const result = await messenger.walletTransaction({
      token: JSON.stringify(payment.token.paymentData),
      amount: 1000, // Amount in cents
      currency: 'USD'
    });
    
    return {
      success: result.state === 'SUCCEEDED',
      transactionId: result.payment_method_id
    };
  } catch (error) {
    console.error('Payment error:', error);
    return { success: false };
  }
}
```

## Complete Example

Here's a minimal working implementation:

```html
<!DOCTYPE html>
<html>
<head>
  <title>Apple Pay Example</title>
  <script src="https://cdn.paytheory.com/messenger/latest/messenger.js"></script>
  <style>
    .apple-pay-button {
      display: none;
      -webkit-appearance: -apple-pay-button;
      -apple-pay-button-type: pay;
      -apple-pay-button-style: black;
      width: 200px;
      height: 44px;
    }
  </style>
</head>
<body>
  <h1>Checkout</h1>
  <p>Total: $10.00</p>
  <button id="apple-pay-button" class="apple-pay-button"></button>
  
  <script>
    const MERCHANT_IDENTIFIER = 'merchant.com.yourdomain.pay';
    const AMOUNT = 10.00;
    
    // Check availability
    if (window.ApplePaySession) {
      ApplePaySession.canMakePaymentsWithActiveCard(MERCHANT_IDENTIFIER)
        .then(canMakePayments => {
          if (canMakePayments) {
            document.getElementById('apple-pay-button').style.display = 'block';
          }
        });
    }
    
    // Handle button click
    document.getElementById('apple-pay-button').addEventListener('click', () => {
      const session = new ApplePaySession(6, {
        countryCode: 'US',
        currencyCode: 'USD',
        supportedNetworks: ['visa', 'masterCard', 'amex', 'discover'],
        merchantCapabilities: ['supports3DS'],
        total: {
          label: 'Your Business Name',
          amount: AMOUNT.toFixed(2)
        }
      });
      
      session.onvalidatemerchant = async (event) => {
        // Call your server for validation
        const response = await fetch('/api/apple-pay/validate', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ validationURL: event.validationURL })
        });
        
        const merchantSession = await response.json();
        session.completeMerchantValidation(merchantSession);
      };
      
      session.onpaymentauthorized = async (event) => {
        // Process with Pay Theory
        const messenger = new PayTheoryMessenger({
          apiKey: 'YOUR_PAY_THEORY_API_KEY'
        });
        
        const result = await messenger.walletTransaction({
          token: JSON.stringify(event.payment.token.paymentData),
          amount: Math.round(AMOUNT * 100),
          currency: 'USD'
        });
        
        if (result.state === 'SUCCEEDED') {
          session.completePayment(ApplePaySession.STATUS_SUCCESS);
          alert('Payment successful!');
        } else {
          session.completePayment(ApplePaySession.STATUS_FAILURE);
          alert('Payment failed');
        }
      };
      
      session.begin();
    });
  </script>
</body>
</html>
```
